function epochMillis(value) {
  if (value == null) {
    throw new Error('timestamp requerido');
  }
  if (value instanceof java.util.Date) {
    return value.getTime();
  }

  var raw = String(value).trim();
  try {
    return java.time.Instant.parse(raw).toEpochMilli();
  } catch (ignored) {
    return java.time.LocalDateTime
      .parse(raw.replace(' ', 'T'))
      .toInstant(java.time.ZoneOffset.UTC)
      .toEpochMilli();
  }
}

for (var i = 0; i < records.length; i++) {
  var record = records[i];
  try {
    var lat2 = Number(record.value['lat']);
    var lon2 = Number(record.value['lon']);
    var currentMs = epochMillis(record.value['event_time']);
    var hasPrevious = record.value['prev_lat'] != null &&
      record.value['prev_lon'] != null &&
      record.value['prev_event_time'] != null;

    var speedKmh = 0.0;

    if (hasPrevious) {
      var lat1 = Number(record.value['prev_lat']);
      var lon1 = Number(record.value['prev_lon']);
      var previousMs = epochMillis(record.value['prev_event_time']);
      var deltaHours = (currentMs - previousMs) / 3600000.0;
      if (deltaHours <= 0.0) {
        throw new Error('event_time debe ser posterior a prev_event_time');
      }

      var dLat = lat2 - lat1;
      var dLon = lon2 - lon1;
      var meanLat = (lat1 + lat2) / 2.0 * Math.PI / 180.0;
      var distanceKm = 111.195 * Math.sqrt(
        dLat * dLat + Math.pow(dLon * Math.cos(meanLat), 2.0)
      );
      speedKmh = distanceKm < 0.05 ? 0.0 : distanceKm / deltaHours;
    }

    var amount = Math.max(0.0, Number(record.value['amount']));

    var speedComponent = speedKmh <= 0.0
      ? 0.0
      : 100.0 * speedKmh / (speedKmh + 80.0);
    var amountComponent = amount <= 0.0
      ? 0.0
      : 80.0 * amount / (amount + 1700000.0);
    var dominant = Math.max(speedComponent, amountComponent);
    var secondary = Math.min(speedComponent, amountComponent);
    var continuousRisk = dominant +
      0.25 * secondary * (1.0 - dominant / 100.0);
    continuousRisk = Math.max(0.0, Math.min(100.0, continuousRisk));

    record.value['fraud_score'] = Math.round(continuousRisk);
    record.value['fraud_category'] =
      speedKmh > 268.224 || continuousRisk >= 70 ? 'FRAUDULENTA' :
      continuousRisk >= 40 ? 'SOSPECHOSA' :
      'NORMAL';

    record.attributes['sdc.operation.type'] = hasPrevious ? '3' : '1';

    output.write(record);
  } catch (exception) {
    error.write(record, String(exception));
  }
}

Bloque 3 — Validar y comprobar

Antes de validar, repasa este checklist rápido:

  • Consumer: Confluent Cloud + entrada TransaccionesEnriquecidas-0 + grupo fraud-workshop-N.
  • Lookup: SQL sobre fraud_customer_gps_N + JSON de correlaciones.
  • Evaluator: script de Pipeline_0 + Lote por lote.
  • Masker: JSON de masking pegado en bulk.
  • Producer: salida TransaccionesEvaluadas-N + subject individual + sin incluir esquema.
  • JDBC Producer: tabla fraud_customer_gps_N + JSON de correlaciones + UPDATE.
  1. Guarda el pipeline completo.
  2. Haz clic en Validar. Corrige cualquier error de conexión o configuración antes de continuar.
  3. Si la validación pasa, inicia Pipeline_N.
  4. En Confluent abre TransaccionesEvaluadas-N y usa Consume latest para ver mensajes recientes.
  • Los mensajes se decodifican como campos Avro.
  • fraud_score está entre 0 y 100.
  • Nombre, correo y tarjeta están enmascarados.
  • La tabla JDBC se actualiza sin errores de primary key.

Problemas frecuentes

SíntomaQué revisar
Sin datos en el tópico de entrada Revisa Confluent + Flink. Avisa al facilitador si el generador no está activo.
Selector de campos vacío en un stage Activa edición en masa y pega el JSON completo de esta guía o de Pipeline_0.
Validación muy lenta Es normal en el workshop. Configura todos los stages y valida una sola vez al final.
Error de conexión JDBC o Kafka Usa la conexión del proyecto. No crees una nueva. Compara con Pipeline_0.
Tópico incorrecto Entrada: TransaccionesEnriquecidas-0. Salida: TransaccionesEvaluadas-N.
Segunda salida del Evaluador sin conectar La rama JDBC sale del Evaluador, no del Enmascarador.
Lookup sin prev_* Revisa la tabla en el SQL y el JSON de correlaciones. Sin historial previo, los campos llegan vacíos (es normal la primera vez).
Mensaje Obj en Confluent En el Productor: Incluir esquema desactivado. Subject: TransaccionesEvaluadas-N-value.
Pipeline en Canceling o engine caído Renueva estado una vez y avisa al facilitador. No lances ejecuciones repetidas.
Error JDBC duplicado Operación UPDATE con customer_id como clave en el Productor de JDBC.
